var currentUrl = ""; 

function render(data) {
    var zone = document.getElementById("list-zone"); var html = ""; window.activeData = data; 
    for (var i = 0; i < data.length; i++) {
        var activeClass = (i === 0) ? "row-news active" : "row-news";
        var imgHtml = data[i].img ? '<img src="' + data[i].img + '" class="img-thumb">' : '';
        html += '<div class="' + activeClass + '" onclick="changeNews(' + i + ', this)">' +
                '<div class="thumb-wrap">' + imgHtml + '</div>' +
                '<div class="txt-wrap"><div class="r-title">' + data[i].title + '</div><div class="r-date">📅 ' + data[i].date + '</div></div>' +
                '</div>';
    }
    zone.innerHTML = html;
    if (data.length > 0) { changeNews(0, document.getElementsByClassName("row-news")[0]); }
}

function changeNews(idx, element) {
    var rows = document.getElementsByClassName("row-news");
    for (var j = 0; j < rows.length; j++) { rows[j].classList.remove("active"); }
    if(element) element.classList.add("active");
    var currentData = window.activeData[idx];
    document.getElementById("p-title").innerHTML = currentData.title;
    document.getElementById("p-desc").innerHTML = "📅 " + currentData.date;
    var pImg = document.getElementById("p-img");
    if (currentData.img) { pImg.src = currentData.img; pImg.style.display = "block"; } else { pImg.style.display = "none"; }
    currentUrl = currentData.url;
}

function goToLink() { if (currentUrl && currentUrl !== "#") { window.open(currentUrl, "_blank"); } }

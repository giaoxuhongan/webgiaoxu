    function cleanText(text) {

        if (!text) return "";

        return text
            .replace(/\s+/g, " ")
            .replace(/\n+/g, " ")
            .trim();
    }


    /* =====================================================

    function absoluteUrl(url) {

        if (!url) return "";

        try {

            return new URL(
                url,
                window.location.origin
            ).href;

        } catch (e) {

            return url;
        }
    }


    /* =====================================================
       10. LẤY URL ẢNH TỪ SRCSET
    ===================================================== */

    function getLargestFromSrcset(srcset) {

        if (!srcset) return "";

        const entries =
            srcset
                .split(",")
                .map(function (part) {

                    const pieces =
                        part.trim()
                            .split(/\s+/);

                    return {

                        url:
                            pieces[0],

                        width:
                            pieces[1]
                                ? parseInt(
                                    pieces[1]
                                        .replace("w", ""),
                                    10
                                )
                                : 0
                    };

                })
                .filter(function (x) {

                    return x.url;
                });


        entries.sort(function (a, b) {

            return b.width - a.width;

        });


        if (entries.length) {

            return absoluteUrl(
                entries[0].url
            );

        }

        return "";
    }


    /* =====================================================

    ===================================================== */

    function getBestImageFromImg(img) {

        if (!img) return "";


        /*
         * Ưu tiên srcset lớn nhất
         */

        const srcset =
            img.getAttribute(
                "srcset"
            ) ||
            img.getAttribute(
                "data-srcset"
            );


        const srcsetImage =
            getLargestFromSrcset(
                srcset
            );


        if (srcsetImage) {

            return srcsetImage;
        }


        /*
         * Ưu tiên data-src / data-original
         */

        const candidates = [

            img.getAttribute(
                "data-original"
            ),

            img.getAttribute(
                "data-full"
            ),

            img.getAttribute(
                "data-large"
            ),

            img.getAttribute(
                "data-src"
            ),

            img.getAttribute(
                "src"
            )

        ];


        for (
            const candidate
            of candidates
        ) {

            if (candidate) {

                return absoluteUrl(
                    candidate
                );
            }
        }


        return "";
    }

    /* =====================================================

    ===================================================== */

    function findDate(element) {

        if (!element) return "";

        /*
         * Chỉ lấy ngày ở vùng hiển thị ngày của từng tin.
         * Không quét toàn bộ textContent của card nữa,
         * vì trong nội dung bài có thể có một ngày khác.
         */

        const dateSelectors = [
            "time",
            ".date",
            ".news-date",
            ".post-date",
            ".article-date",
            ".item-date",
            ".entry-date",
            ".created-date",
            ".date-posted",
            "[class*='date']"
        ];

        for (const selector of dateSelectors) {

            const nodes =
                Array.from(
                    element.querySelectorAll(selector)
                );

            for (const node of nodes) {

                const text =
                    cleanText(
                        node.textContent
                    );

                const match =
                    text.match(
                        /(?:Ngày\s*:?\s*)?(\d{1,2}[-\/]\d{1,2}[-\/]\d{4})/i
                    );

                if (match) {

                    return "Ngày: " +
                        match[1];
                }
            }
        }

        /*
         * Nếu website không đặt class cho ngày,
         * tìm các phần tử nhỏ có nội dung gần như chỉ là ngày.
         * Không quét các khối lớn chứa cả nội dung bài.
         */

        const smallNodes =
            Array.from(
                element.querySelectorAll(
                    "span, small, time, p, div"
                )
            );

        for (const node of smallNodes) {

            const text =
                cleanText(
                    node.textContent
                );

            if (
                text.length > 40 ||
                text.length < 8
            ) {
                continue;
            }

            const match =
                text.match(
                    /^(?:Ngày\s*:?\s*)?(\d{1,2}[-\/]\d{1,2}[-\/]\d{4})$/i
                );

            if (match) {

                return "Ngày: " +
                    match[1];
            }
        }

        return "";
    }


    /* =====================================================

    function findContentArea(doc, categoryName) {

        const wanted = cleanText(categoryName).toLowerCase();
        const headings = Array.from(doc.querySelectorAll("h1,h2,h3,h4,h5,h6"));

        /* Ưu tiên heading trùng tên tab */
        for (const heading of headings) {
            const text = cleanText(heading.textContent).toLowerCase();
            if (!(text === wanted || text.includes(wanted) || wanted.includes(text))) continue;

            let area = heading.parentElement;
            for (let i = 0; i < 6 && area; i++) {
                if (area.querySelectorAll("a[href]").length >= 4) return area;
                area = area.parentElement;
            }
        }

        /* Thử khớp các từ chính của tên tab */
        const words = wanted.split(/\s+/).filter(function(word) {
            return word.length >= 4;
        });

        for (const heading of headings) {
            const text = cleanText(heading.textContent).toLowerCase();
            const hits = words.filter(function(word) {
                return text.includes(word);
            }).length;

            if (hits < Math.min(2, words.length)) continue;

            let area = heading.parentElement;
            for (let i = 0; i < 6 && area; i++) {
                if (area.querySelectorAll("a[href]").length >= 4) return area;
                area = area.parentElement;
            }
        }

        /* Không tìm thấy vùng rõ ràng thì tuyệt đối không quét document.body */
        const safeSelectors = [
            "main", "[role='main']", "article",
            ".news-list", ".post-list", ".blog-list",
            ".article-list", ".item-list", ".news-container",
            ".content-area", ".main-content"
        ];

        for (const selector of safeSelectors) {
            const area = doc.querySelector(selector);
            if (area && area.querySelectorAll("a[href]").length >= 4) return area;
        }

        return null;
    }

    /* =====================================================
       15. PHÂN TÍCH DANH MỤC
    ===================================================== */

    function parseCategory(
        html,
        category
    ) {

        const parser =
            new DOMParser();


        const doc =
            parser.parseFromString(
                html,
                "text/html"
            );


        const area =
            findContentArea(
                doc,
                category.name
            );


        if (!area) {
            return [];
        }

        const links =
            Array.from(
                area.querySelectorAll(
                    "a[href]"
                )
            );


        const result = [];

        const used =
            new Set();


        for (
            const link
            of links
        ) {

            const href =
                link.getAttribute(
                    "href"
                );


            if (!href) continue;


            if (
                href === "#" ||
                href.startsWith(
                    "javascript:"
                )
            ) continue;


            const title =
                cleanText(
                    link.textContent
                );


            if (
                title.length < 8
            ) continue;


            const url =
                absoluteUrl(
                    href
                );


            if (
                !url ||
                used.has(url)
            ) continue;


            /*
             * Bỏ các link menu
             */

            const lower =
                title.toLowerCase();


            /*
             * BỎ LINK TÊN DANH MỤC
             *
             * Một số chuyên mục có một link đầu tiên mang đúng
             * tên danh mục (ví dụ: "Lời Chúa mỗi ngày").
             * Link này không phải là một bài viết nên không được
             * đưa vào danh sách tin.
             */
            const categoryName =
                cleanText(
                    category.name
                ).toLowerCase();


            if (
                categoryName &&
                lower === categoryName
            ) {

                continue;
            }


            const bad =
                [
                    "giới thiệu",
                    "giáo họ",
                    "hội đoàn",
                    "liên hệ",
                    "tài liệu",
                    "liên kết",
                    "trang chủ",
                    "đăng nhập",
                    "thoát"
                ];


            if (
                bad.some(function (word) {

                    return lower === word;

                })
            ) {

                continue;
            }


            /*
             * Tìm card cha
             */

            let card =
                link.parentElement;


            let image = "";

            let date = "";

            let description = "";


            for (
                let i = 0;
                i < 6;
                i++
            ) {

                if (!card) break;


                if (!image) {

                    image =
                        findCardImage(
                            card
                        );
                }


                if (!date) {

                    date =
                        findDate(
                            card
                        );
                }


                const text =
                    cleanText(
                        card.textContent
                    );


                if (
                    text.length > 30
                ) {

                    description =
                        text;
                }


                if (
                    image &&
                    date
                ) {

                    break;
                }


                card =
                    card.parentElement;
            }


            /*
             * Bỏ những link không phải bài
             */

            if (
                lower.includes(
                    "tin giáo phận"
                ) ||
                lower.includes(
                    "tin giáo xứ"
                )
            ) {

                continue;
            }


            description =
                description
                    .replace(
                        title,
                        ""
                    )
                    .replace(
                        date,
                        ""
                    )
                    .trim();


            result.push({

                title:
                    title,

                url:
                    url,

                image:
                    image,

                date:
                    date,

                description:
                    description

            });


            used.add(url);


            if (
                result.length >=
                YVB_V4_CONFIG.maxItems
            ) {

                break;
            }
        }


        return result;
    }


    /* =====================================================
       16. LẤY OG IMAGE TỪ TRANG BÀI
    ===================================================== */

    function findMetaImage(doc) {

        const selectors = [

            'meta[property="og:image"]',

            'meta[property="og:image:url"]',

            'meta[name="twitter:image"]',

            'meta[name="twitter:image:src"]'

        ];


        for (
            const selector
            of selectors
        ) {

            const meta =
                doc.querySelector(
                    selector
                );


            if (meta) {

                const content =
                    meta.getAttribute(
                        "content"
                    );


                if (content) {

                    return absoluteUrl(
                        content
                    );
                }
            }
        }


        return "";
    }


    /* =====================================================
       17. TÌM ẢNH LỚN NHẤT TRONG BÀI
    ===================================================== */

    function findLargestArticleImage(
        doc
    ) {

        const images =
            Array.from(
                doc.querySelectorAll(
                    "img"
                )
            );


        const candidates = [];


        for (
            const img
            of images
        ) {

            const src =
                getBestImageFromImg(
                    img
                );


            if (!src) continue;


            /*
             * Bỏ ảnh logo / icon / menu
             */

            const alt =
                (
                    img.getAttribute(
                        "alt"
                    ) || ""
                ).toLowerCase();


            const cls =
                (
                    img.className || ""
                ).toString()
                    .toLowerCase();


            if (
                alt.includes(
                    "logo"
                ) ||
                cls.includes(
                    "logo"
                ) ||
                cls.includes(
                    "icon"
                )
            ) {

                continue;
            }


            /*
             * Ưu tiên ảnh có kích thước
             * HTML lớn.
             */

            const width =
                parseInt(
                    img.getAttribute(
                        "width"
                    ) || "0",
                    10
                );


            const height =
                parseInt(
                    img.getAttribute(
                        "height"
                    ) || "0",
                    10
                );


            const area =
                width * height;


            candidates.push({

                url:
                    src,

                area:
                    area

            });
        }


        candidates.sort(
            function (a, b) {

                return b.area - a.area;

            }
        );


        if (
            candidates.length
        ) {

            return candidates[0].url;
        }


        return "";
    }


    /* =====================================================
       18. LẤY ẢNH GỐC CỦA BÀI
    ===================================================== */

    async function getArticleImage(
        item
    ) {

        /*
         * Trước tiên dùng ảnh thumbnail
         * để người dùng không phải chờ.
         */

        let fallback =
            item.image || "";


        try {

            const response =
                await fetch(
                    item.url,
                    {
                        method: "GET",

                        credentials:
                            "same-origin",

                        cache:
                            "no-store",

                        headers: {
                            "Accept":
                                "text/html"
                        }
                    }
                );


            if (
                !response.ok
            ) {

                return fallback;
            }


            const html =
                await response.text();


            const parser =
                new DOMParser();


            const doc =
                parser.parseFromString(
                    html,
                    "text/html"
                );


            /*
             * Ưu tiên OG IMAGE
             */

            const ogImage =
                findMetaImage(
                    doc
                );


            if (ogImage) {

                return ogImage;
            }


            /*
             * Nếu không có OG image,
             * tìm ảnh lớn nhất trong bài.
             */

            const articleImage =
                findLargestArticleImage(
                    doc
                );


            if (articleImage) {

                return articleImage;
            }


        } catch (error) {

            console.warn(
                "YVB V3 - không lấy được ảnh gốc:",
                error
            );
        }


        return fallback;
    }


    /* =====================================================
       19. HIỂN THỊ ẢNH
    ===================================================== */

    function displayImage(
        url,
        title
    ) {

        imageLoading.style.display =
            "block";


        mainImage.style.opacity =
            "0";


        mainImage.removeAttribute(
            "src"
        );


        if (!url) {

            imageLoading.textContent =
                "Không có ảnh";

            imageLoading.style.display =
                "block";

            return;
        }


        const img =
            new Image();


        img.onload =
            function () {

                mainImage.src =
                    url;

                mainImage.alt =
                    title || "";

                mainImage.style.opacity =
                    "1";

                imageLoading.style.display =
                    "none";
            };


        img.onerror =
            function () {

                imageLoading.textContent =
                    "Không tải được ảnh";

                imageLoading.style.display =
                    "block";

                mainImage.style.opacity =
                    "0";
            };


        img.src =
            url;
    }


    /* =====================================================
       20. HIỂN THỊ PREVIEW
    ===================================================== */

    async function showPreview(
        item,
        element
    ) {

        if (!item) return;


        selectedItem =
            item;


        /*
         * Active item
         */

        shadow
            .querySelectorAll(
                ".yvb4-item"
            )
            .forEach(
                function (el) {

                    el.classList.toggle(
                        "selected",
                        el === element
                    );

                }
            );


        previewTitle.textContent =
            item.title;


        previewDate.textContent =
            item.date || "";


        previewDescription.textContent =
            item.description || "";


        detail.href =
            item.url || "#";


        /*
         * Hiển thị thumbnail ngay
         */

        displayImage(
            item.image,
            item.title
        );


        /*
         * Sau đó lấy ảnh gốc
         */

        const bestImage =
            await getArticleImage(
                item
            );


        /*
         * Nếu người dùng đã chuyển sang
         * bài khác trong lúc đang tải,
         * không thay ảnh cũ.
         */

        if (
            selectedItem !== item
        ) {

            return;
        }


        if (
            bestImage &&
            bestImage !==
            item.image
        ) {

            displayImage(
                bestImage,
                item.title
            );
        }
    }

       21. HIỂN THỊ DANH SÁCH
    ===================================================== */

    function renderList(
        items
    ) {

        list.innerHTML = "";


        if (
            !items.length
        ) {

            list.innerHTML = `

                <div class="yvb4-error">

                    Không lấy được danh sách bài viết.

                </div>

            `;

            return;
        }


        items.forEach(
            function (
                item,
                index
            ) {

                const article =
                    document.createElement(
                        "article"
                    );


                article.className =
                    "yvb4-item";


                const thumb =
                    document.createElement(
                        "img"
                    );


                thumb.className =
                    "yvb4-thumb";


                thumb.alt =
                    item.title || "";


                if (
                    item.image
                ) {

                    thumb.src =
                        item.image;
                }


                const info =
                    document.createElement(
                        "div"
                    );


                info.className =
                    "yvb4-item-info";


                const title =
                    document.createElement(
                        "div"
                    );


                title.className =
                    "yvb4-item-title";


                title.textContent =
                    item.title;


                const date =
                    document.createElement(
                        "div"
                    );


                date.className =
                    "yvb4-item-date";


                date.textContent =
                    item.date || "";


                info.appendChild(
                    title
                );


                info.appendChild(
                    date
                );


                article.appendChild(
                    thumb
                );


                article.appendChild(
                    info
                );


                article.addEventListener(
                    "click",
                    function () {

                        showPreview(
                            item,
                            article
                        );

                    }
                );

                article.addEventListener(
                    "dblclick",
                    function (event) {

                        event.preventDefault();

                        if (item.url) {
                            window.location.href = item.url;
                        }

                    }
                );


                list.appendChild(
                    article
                );


                /*
                 * Bài đầu tiên tự động chọn
                 */

                if (
                    index === 0
                ) {

                    setTimeout(
                        function () {

                            showPreview(
                                item,
                                article
                            );

                        },
                        50
                    );
                }

            }
        );


        /*
         * Nút xem thêm
         */

        const more =
            document.createElement(
                "a"
            );


        more.className =
            "yvb4-more";


        more.href =
            YVB_V4_CONFIG
                .categories[
                    currentCategory
                ]
                .url;


        more.target =
            "_blank";


        more.rel =
            "noopener noreferrer";


        more.textContent =
            "XEM THÊM TIN";


        list.appendChild(
            more
        );
    }

       22. TẢI DANH MỤC
    ===================================================== */

    async function loadCategory(
        category
    ) {

        try {

            const response =
                await fetch(
                    category.url,
                    {
                        method:
                            "GET",

                        credentials:
                            "same-origin",

                        cache:
                            "no-store",

                        headers: {
                            "Accept":
                                "text/html"
                        }
                    }
                );


            if (
                !response.ok
            ) {

                throw new Error(
                    "HTTP " +
                    response.status
                );
            }


            const html =
                await response.text();


            return parseCategory(
                html,
                category
            );


        } catch (error) {

            console.error(
                "YVB NewsBox V2:",
                error
            );


            return [];
        }
    }

    ) {

         */

                ".yvb4-tab"
            )

                    i
                ) {

                    tab.classList.toggle(

                    );

            );

            YVB_V4_CONFIG

            category.name;


        list.innerHTML = `

            <div class="yvb4-loading">

                Đang tải danh sách tin...

            </div>

        `;


        /*
         * Reset preview
         */

        selectedItem =
            null;


        const items =
            await loadCategory(
                category
            );


        renderList(
            items
        );
    }

var currentUrl = "";

// Hàm tự động tạo danh sách tin ở dưới
function renderList(data) {
    var zone = document.getElementById("sub-list-zone");
    var html = "";
    
    for (var i = 0; i < data.length; i++) {
        var activeClass = (i === 0) ? "yvb4-item selected" : "yvb4-item";
        var imgHtml = data[i].img ? '<img src="' + data[i].img + '" class="yvb4-thumb">' : '';
        
        html += '<div class="' + activeClass + '" onclick="selectNews(' + i + ', this)">' +
                imgHtml +
                '<div class="yvb4-item-info">' +
                    '<div class="yvb4-item-title">' + data[i].title + '</div>' +
                    '<div class="yvb4-item-date">📅 ' + data[i].date + '</div>' +
                '</div>' +
                '</div>';
    }
    zone.innerHTML = html;
    
    // Mặc định hiển thị bài đầu tiên lên màn hình xem trước phía trên
    if (data.length > 0) {
        updatePreview(data[0]);
    }
}

// Hàm cập nhật màn hình xem trước phía trên khi chọn tin
function selectNews(index, element) {
    var items = document.getElementsByClassName("yvb4-item");
    for (var j = 0; j < items.length; j++) {
        items[j].classList.remove("selected");
    }
    element.classList.add("selected");
    
    updatePreview(DANH_SACH_TIN[index]);
}

function updatePreview(newsItem) {
    document.getElementById("main-title").innerHTML = newsItem.title;
    document.getElementById("main-date").innerHTML = "📅 Ngày đăng: " + newsItem.date;
    document.getElementById("main-desc").innerHTML = newsItem.desc;
    document.getElementById("main-link").href = newsItem.url;
    
    var mainImg = document.getElementById("main-img");
    if (newsItem.img && newsItem.img !== "") {
        mainImg.src = newsItem.img;
        mainImg.parentElement.style.display = "flex";
    } else {
        mainImg.parentElement.style.display = "none";
    }
    currentUrl = newsItem.url;
}

// Lệnh click mở tab mới khi bấm vào ảnh hoặc tiêu đề lớn phía trên
function goToLink() {
    if (currentUrl && currentUrl !== "#") {
        window.open(currentUrl, "_blank");
    }
}

// Chạy khởi tạo danh sách tin khi mở trang
